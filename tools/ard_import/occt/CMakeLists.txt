cmake_minimum_required(VERSION 3.21)
project(ArdSnapshotToOcct LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(OpenCASCADE REQUIRED)
add_executable(ard_snapshot_to_occt geometry_to_occt.cpp)
target_include_directories(ard_snapshot_to_occt PRIVATE ${OpenCASCADE_INCLUDE_DIR})
if(TARGET TKDESTEP)
  set(STEP_LIB TKDESTEP)
else()
  set(STEP_LIB TKSTEP)
endif()
target_link_libraries(ard_snapshot_to_occt PRIVATE Qt6::Core ${STEP_LIB} TKShHealing TKTopAlgo TKBRep TKG3d TKGeomBase TKMath TKernel)
